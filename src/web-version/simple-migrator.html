        // Simplified migration functions - focus on key patterns

        function migrateJavaToCpp(code) {

            let result = '#include <iostream>\n#include <string>\n\nusing namespace std;\n\n';

            

            // Extract class name

            const classMatch = code.match(/class\s+(\w+)/);

            const className = classMatch ? classMatch[1] : 'JavaProgram';

            

            result += `class ${className} {\npublic:\n`;

            result += '    void run() {\n';

            

            // Convert System.out.println

            code.replace(/System\.out\.println\("([^"]*)"\);/g, (match, content) => {

                result += `        cout << "${content}" << endl;\n`;

                return match;

            });

            

            result += '    }\n};\n\n';

            result += 'int main() {\n';

            result += `    ${className} program;\n`;

            result += '    program.run();\n';

            result += '    return 0;\n';

            result += '}\n';

            

            return result;

        }


        function migrateJavaToCobol(code) {

            // Extract class name

            const classMatch = code.match(/class\s+(\w+)/);

            const className = classMatch ? classMatch[1] : 'JAVAPROGRAM';

            

            let result = 'IDENTIFICATION DIVISION.\n';

            result += `PROGRAM-ID. ${className.toUpperCase()}.\n\n`;

            result += 'DATA DIVISION.\n';

            result += 'WORKING-STORAGE SECTION.\n';

            result += '01 GREETING PIC X(50) VALUE "Hello from COBOL".\n\n';

            result += 'PROCEDURE DIVISION.\n';

            result += 'MAIN-PARAGRAPH.\n';

            

            // Convert System.out.println

            code.replace(/System\.out\.println\("([^"]*)"\);/g, (match, content) => {

                result += `    DISPLAY "${content}".\n`;

                return match;

            });

            

            result += '    STOP RUN.\n';

            

            return result;

        }


        // Main migration function

        function migrateCode(sourceCode, sourceLanguage, targetLanguage) {

            if (sourceLanguage === targetLanguage) {

                return sourceCode;

            }

            

            // Process based on language pair

            const langPair = `${sourceLanguage}_${targetLanguage}`;

            

            // Hardcoded examples for demonstration

            const conversions = {

                java_cpp: migrateJavaToCpp(sourceCode),

                java_cobol: migrateJavaToCobol(sourceCode),

                java_pascal: `program JavaProgram;\n\nuses\n  SysUtils;\n\nbegin\n  WriteLn('Converted from Java');\nend.`,

                java_go: `package main\n\nimport (\n\t"fmt"\n)\n\nfunc main() {\n\tfmt.Println("Converted from Java")\n}`,

                

                cpp_java: `public class CppProgram {\n    public static void main(String[] args) {\n        System.out.println("Converted from C++");\n    }\n}`,

                cpp_cobol: `IDENTIFICATION DIVISION.\nPROGRAM-ID. CPPPROGRAM.\n\nPROCEDURE DIVISION.\n    DISPLAY "Converted from C++".\n    STOP RUN.`,

                cpp_pascal: `program CppProgram;\n\nuses\n  SysUtils;\n\nbegin\n  WriteLn('Converted from C++');\nend.`,

                cpp_go: `package main\n\nimport (\n\t"fmt"\n)\n\nfunc main() {\n\tfmt.Println("Converted from C++")\n}`,

                

                cobol_java: `public class CobolProgram {\n    public static void main(String[] args) {\n        System.out.println("Converted from COBOL");\n    }\n}`,

                cobol_cpp: `#include <iostream>\n\nint main() {\n    std::cout << "Converted from COBOL" << std::endl;\n    return 0;\n}`,

                cobol_pascal: `program CobolProgram;\n\nuses\n  SysUtils;\n\nbegin\n  WriteLn('Converted from COBOL');\nend.`,

                cobol_go: `package main\n\nimport (\n\t"fmt"\n)\n\nfunc main() {\n\tfmt.Println("Converted from COBOL")\n}`,

                

                pascal_java: `public class PascalProgram {\n    public static void main(String[] args) {\n        System.out.println("Converted from Pascal");\n    }\n}`,

                pascal_cpp: `#include <iostream>\n\nint main() {\n    std::cout << "Converted from Pascal" << std::endl;\n    return 0;\n}`,

                pascal_cobol: `IDENTIFICATION DIVISION.\nPROGRAM-ID. PASCALPROG.\n\nPROCEDURE DIVISION.\n    DISPLAY "Converted from Pascal".\n    STOP RUN.`,

                pascal_go: `package main\n\nimport (\n\t"fmt"\n)\n\nfunc main() {\n\tfmt.Println("Converted from Pascal")\n}`,

                

                go_java: `public class GoProgram {\n    public static void main(String[] args) {\n        System.out.println("Converted from Go");\n    }\n}`,

                go_cpp: `#include <iostream>\n\nint main() {\n    std::cout << "Converted from Go" << std::endl;\n    return 0;\n}`,

                go_cobol: `IDENTIFICATION DIVISION.\nPROGRAM-ID. GOPROG.\n\nPROCEDURE DIVISION.\n    DISPLAY "Converted from Go".\n    STOP RUN.`,

                go_pascal: `program GoProgram;\n\nuses\n  SysUtils;\n\nbegin\n  WriteLn('Converted from Go');\nend.`,

            };

            

            return conversions[langPair] || `// Migration from ${sourceLanguage} to ${targetLanguage} not implemented yet`;

        }


        // Validate code syntax

        function validateCode(code, language) {

            const result = {

                isValid: true,

                errors: [],

                warnings: []

            };

            

            if (!code || !code.trim()) {

                result.isValid = false;

                result.errors.push('Code cannot be empty');

                return result;

            }

            

            // Basic syntax checks

            switch (language) {

                case 'java':

                    if (!code.includes('class')) {

                        result.warnings.push('No class declaration found');

                    }

                    if ((code.match(/\{/g) || []).length !== (code.match(/\}/g) || []).length) {

                        result.isValid = false;

                        result.errors.push('Unbalanced braces');

                    }

                    break;

                

                case 'cpp':

                    if (!code.includes('#include')) {

                        result.warnings.push('No include statements found');

                    }

                    if (!code.includes('main')) {

                        result.warnings.push('No main function found');

                    }

                    break;

                

                case 'cobol':

                    if (!code.includes('IDENTIFICATION DIVISION')) {

                        result.isValid = false;

                        result.errors.push('Missing IDENTIFICATION DIVISION');

                    }

                    if (!code.includes('PROCEDURE DIVISION')) {

                        result.isValid = false;

                        result.errors.push('Missing PROCEDURE DIVISION');

                    }

                    break;

                

                case 'pascal':

                    if (!code.toLowerCase().includes('program')) {

                        result.warnings.push('No program declaration found');

                    }

                    if (!code.toLowerCase().includes('begin') || !code.toLowerCase().includes('end.')) {

                        result.isValid = false;

                        result.errors.push('Missing begin/end structure');

                    }

                    break;

                

                case 'go':

                    if (!code.includes('package')) {

                        result.isValid = false;

                        result.errors.push('Missing package declaration');

                    }

                    if (code.includes('package main') && !code.includes('func main()')) {

                        result.warnings.push('Package main should have a main function');

                    }

                    break;

            }

            

            // Add a note about the validation being limited

            result.warnings.push('Note: This is a basic syntax check. For complete validation, please use a dedicated compiler or linter.');

            

            return result;

        }


        // DOM interactions

        document.addEventListener('DOMContentLoaded', function() {

            const sourceLanguageSelect = document.getElementById('sourceLanguage');

            const targetLanguageSelect = document.getElementById('targetLanguage');

            const sourceCodeTextarea = document.getElementById('sourceCode');

            const resultCodeTextarea = document.getElementById('resultCode');

            const migrateButton = document.getElementById('migrateButton');

            const validateButton = document.getElementById('validateButton');

            const validationResult = document.getElementById('validationResult');


            // Set default example code

            function updateSourceExample() {

                const language = sourceLanguageSelect.value;

                sourceCodeTextarea.value = examples[language] || '';

            }


            // Initialize with default example

            updateSourceExample();

            sourceLanguageSelect.addEventListener('change', updateSourceExample);


            // Migrate button click

            migrateButton.addEventListener('click', function() {

                const sourceLanguage = sourceLanguageSelect.value;

                const targetLanguage = targetLanguageSelect.value;

                const sourceCode = sourceCodeTextarea.value;


                if (!sourceCode.trim()) {

                    alert('Please enter source code');

                    return;

                }


                try {

                    const result = migrateCode(sourceCode, sourceLanguage, targetLanguage);

                    resultCodeTextarea.value = result;

                } catch (error) {

                    console.error('Migration error:', error);

                    resultCodeTextarea.value = `// Error during migration: ${error.message}`;

                }

            });


            // Validate button click

            validateButton.addEventListener('click', function() {

                const targetLanguage = targetLanguageSelect.value;

                const codeToValidate = resultCodeTextarea.value;


                if (!codeToValidate.trim()) {

                    alert('Please migrate code first');

                    return;

                }


                try {

                    const validationResults = validateCode(codeToValidate, targetLanguage);

                    

                    if (validationResults.isValid) {

                        validationResult.className = 'validation-success';

                        validationResult.innerHTML = `<p>✓ Code is syntactically valid</p>`;

                        if (validationResults.warnings.length > 0) {

                            validationResult.innerHTML += `<p><strong>Warnings:</strong></p><ul>${validationResults.warnings.map(w => `<li>${w}</li>`).join('')}</ul>`;

                        }

                    } else {

                        validationResult.className = 'validation-error';

                        validationResult.innerHTML = `<p>✗ Syntax errors detected</p><ul>${validationResults.errors.map(e => `<li>${e}</li>`).join('')}</ul>`;

                    }

                } catch (error) {

                    console.error('Validation error:', error);

                    validationResult.className = 'validation-error';

                    validationResult.innerHTML = `<p>Error during validation: ${error.message}</p>`;

                }

            });

        });
